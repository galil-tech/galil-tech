🐍 robot7.py – הילדים לוחצים F5

עם "התוכנה האחת": LESSON = 7. עד אז – הקובץ הזה במחשב של כל צוות:

import time
from soil_moisture_driver import SoilMoistureSensor
from touch_driver import TouchSensor
from ws2812_driver import WS2812LED
from buzzer_driver import PassiveBuzzer

TON = 1000   # the robot's voice: 400 - 2000

soil = SoilMoistureSensor(pin_num=6)
touch = TouchSensor(pin_num=4)
led = WS2812LED(pin_num=48, brightness=0.2)
buzzer = PassiveBuzzer(pin_num=5)

def wait_for_touch():
    while touch.is_touched():
        time.sleep_ms(50)
    while not touch.is_touched():
        time.sleep_ms(50)
    buzzer.alarm_beep(count=1, freq=1000, beep_ms=80)
    while touch.is_touched():
        time.sleep_ms(50)
    time.sleep_ms(300)

led.red()             # 1. sensor in AIR - touch
wait_for_touch()
soil.calibrate_dry()
led.green()           # 2. sensor in WATER - touch
wait_for_touch()
soil.calibrate_wet()

last_alarm = 0
while True:
    if soil.read_percent() < 30:      # dry -> shout
        led.red()
        if time.time() - last_alarm >= 3:
            buzzer.alarm_beep(count=3, freq=TON, beep_ms=100)
            last_alarm = time.time()
    else:
        led.green()
    time.sleep(1)
