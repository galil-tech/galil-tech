🐍 הקוד של היום – הילדים רק לוחצים F5

עם "התוכנה האחת": LESSON = 6, TRACK = "elem". עד אז: שמרו את הקובץ הזה במחשב של כל צוות בשם robot6.py ופתחו אותו ב-Thonny לפני השיעור:

import time
from soil_moisture_driver import SoilMoistureSensor
from touch_driver import TouchSensor
soil = SoilMoistureSensor(pin_num=6)
touch = TouchSensor(pin_num=4)   # don't touch while starting!
while True:
    number = round(soil.read_raw() * 100 / 65535)   # 0-100, high = dry
    if touch.is_touched():
        print(number, "  TOUCH!")
    else:
        print(number)
    time.sleep(1)

ההדפסות באנגלית בכוונה – עברית מתהפכת בחלון של Thonny. "TOUCH!" = "נגעו בי!".
