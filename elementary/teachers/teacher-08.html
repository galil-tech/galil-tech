🐍 robot8.py – הילדים לוחצים F5

עם "התוכנה האחת": LESSON = 8, TRACK = "elem". במסך: שם הרובוט + מילה במקום מספר – THIRSTY / OK / HAPPY (כי ביסודי "גבוה = יבש", ואחוז לחות היה מבלבל). הצוות משנה ROBOT_NAME.

עד אז – המעגל בלי מסך ▾ 
import time
from soil_moisture_driver import SoilMoistureSensor
from touch_driver import TouchSensor
from ws2812_driver import WS2812LED
from buzzer_driver import PassiveBuzzer

soil = SoilMoistureSensor(pin_num=6)
touch = TouchSensor(pin_num=4)
led = WS2812LED(pin_num=48, brightness=0.2)
buzzer = PassiveBuzzer(pin_num=5)

def wait_for_touch():
    while touch.is_touched():
        time.sleep_ms(50)
    while not touch.is_touched():
        time.sleep_ms(50)
    buzzer.alarm_beep(count=1, freq=1000, beep_ms=80)
    while touch.is_touched():
        time.sleep_ms(50)
    time.sleep_ms(300)

led.red();   wait_for_touch(); soil.calibrate_dry()   # AIR
led.green(); wait_for_touch(); soil.calibrate_wet()   # WATER

last_alarm = 0
while True:
    moisture = soil.read_percent()
    if moisture < 30:              # thirsty
        led.red()
        if time.time() - last_alarm >= 3:
            buzzer.alarm_beep(count=3, freq=1000, beep_ms=100)
            last_alarm = time.time()
    elif moisture <= 60:           # a little thirsty
        led.yellow()
    else:                          # happy
        led.green()
    time.sleep(1)
