📦 קופסת המסתורין (ערכת המדריך/ה)

import time
from touch_driver import TouchSensor
from ws2812_driver import WS2812LED
from buzzer_driver import PassiveBuzzer
touch = TouchSensor(pin_num=4)
led = WS2812LED(pin_num=48, brightness=0.2)
buzzer = PassiveBuzzer(pin_num=5)
while True:
    if touch.is_touched():
        led.blue()
        buzzer.alarm_beep(count=2, freq=1000, beep_ms=100)
        time.sleep(1)
    else:
        led.off()
    time.sleep_ms(50)

🐍 קוד הרובוט (F5) – SAF ו-TON הם "השינוי האחד"

import time
from soil_moisture_driver import SoilMoistureSensor
from ws2812_driver import WS2812LED
from buzzer_driver import PassiveBuzzer

SAF = 30     # below this % the robot gets "stressed"
TON = 1000   # the robot's voice (Hz)

soil = SoilMoistureSensor(pin_num=6)
led = WS2812LED(pin_num=48, brightness=0.2)
buzzer = PassiveBuzzer(pin_num=5)

input("1. Sensor in AIR, press Enter")
soil.calibrate_dry()
input("2. Sensor in WATER, press Enter")
soil.calibrate_wet()

last_alarm = 0
while True:
    moisture = soil.read_percent()
    print("Moisture:", moisture, "%")
    if moisture < SAF:
        led.red()
        if time.time() - last_alarm >= 3:   # not every second
            buzzer.alarm_beep(count=3, freq=TON, beep_ms=100)
            last_alarm = time.time()
    else:
        led.green()
    time.sleep(1)

כשתגיע "התוכנה האחת": LESSON = 7 – הכיול שם במגע במקום Enter.
